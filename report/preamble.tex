%%%%% Theorem-like environments %%%%%
% \begin{thm} ... \end{thm}, with two optional switches before the title:
%   *  start the body on a new line;   !  draw the statement in a blue box.
% e.g. \begin{thm}*![Exact dimer ground state]. Theorems, lemmas, propositions and
% corollaries share one counter, as do exercises and problems.
\theoremstyle{definition}
\tcbset{kvthmbox/.style={breakable, enhanced, colback=blue!5!white,
  colframe=blue!75!black, fonttitle=\bfseries}}
\makeatletter
% \kvnewtheorem[<shared env>]{<env>}{<Name>}; the starred form numbers globally.
\NewDocumentCommand{\kvnewtheorem}{s o m m}{%
  \IfNoValueTF{#2}
    {\IfBooleanTF{#1}{\newtheorem{#3plain}{#4}}{\newtheorem{#3plain}{#4}[section]}%
     \expandafter\def\csname kv@ctr@#3\endcsname{#3plain}}
    {\newtheorem{#3plain}[#2plain]{#4}%
     \expandafter\def\csname kv@ctr@#3\endcsname{#2plain}}%
  \NewDocumentEnvironment{#3}{t* t! o}
    {\IfBooleanTF{##2}
      {\refstepcounter{\csname kv@ctr@#3\endcsname}%
       \begin{tcolorbox}[kvthmbox, title={#4~\csname the\csname kv@ctr@#3\endcsname
         \endcsname\IfNoValueF{##3}{ (##3)}}]}
      {\IfNoValueTF{##3}{\begin{#3plain}}{\begin{#3plain}[##3]}}%
     \IfBooleanT{##1}{\leavevmode\par}}
    {\IfBooleanTF{##2}{\end{tcolorbox}}{\end{#3plain}}}%
}
\makeatother

% \cref{prefix:name}: "Theorem 3" etc. from the label prefix (cleveref does not see the
% shared theorem counters, so the name comes from our label convention instead).
\ExplSyntaxOn
\seq_new:N \l__kv_prefix_seq
\NewDocumentCommand{\cref}{m}
  {
    \regex_extract_once:nnN { \A[a-z]+ } {#1} \l__kv_prefix_seq
    \str_case_e:nnF { \seq_item:Nn \l__kv_prefix_seq {1} }
      {
        {thm}{Theorem} {lem}{Lemma} {prop}{Proposition} {cor}{Corollary}
        {dfn}{Definition} {rem}{Remark} {sec}{Section} {fig}{Figure} {tab}{Table}
        {exer}{Exercise} {eq}{Equation} {app}{Appendix} {alg}{Algorithm}
      } { }
    \nobreakspace\ref{#1}
  }
\NewCommandCopy{\Cref}{\cref}
\ExplSyntaxOff
